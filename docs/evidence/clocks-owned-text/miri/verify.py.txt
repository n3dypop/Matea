"""Bind targeted interpreted tests to their actual committed compile inputs."""
import hashlib,json,subprocess
from pathlib import Path
if not __debug__:raise RuntimeError('verification refuses -O')
root=Path.cwd();base=root/'docs/evidence/clocks-owned-text/miri';sha=lambda b:hashlib.sha256(b).hexdigest()
context=json.loads((base/'context.json').read_bytes());assert context['status']=='complete' and context['seed']==223 and len(context['commands'])==3
for command in context['commands']:
 assert command['exit_code']==0 and sha((base/command['log']).read_bytes())==command['log_sha256']
 assert 'test result: ok. 1 passed; 0 failed;' in (base/command['log']).read_text()
for name,h in context['source_sha256'].items():assert sha(subprocess.check_output(['git','show',context['source_head']+':'+name]))==h,name
assert 'procedure_sha256' not in context  # Original journal limitation is retained.
result={'schema':1,'status':'complete','source_head':context['source_head'],'targeted_tests':3,'seed':223,'compile_test_inputs':len(context['source_sha256']),
 'producer_procedure_sha256_at_verification':sha((base/'procedure.py.txt').read_bytes()),'context_sha256':sha((base/'context.json').read_bytes()),'procedure_sha256':sha(Path(__file__).read_bytes()),
 'limits':context['limits']+['The original journal did not record a producer-script hash. Its retained script is hashed at verification time only; this is not an execution-time observation.']}
(base/'verification.json').write_text(json.dumps(result,indent=2)+'\n');print('Verified three Miri tests and',len(context['source_sha256']),'actual committed compile/test inputs')
