import hashlib,json,re,statistics,subprocess,sys,tarfile
from functools import lru_cache

ROOT=Path.cwd();BASE=ROOT/'docs/evidence/auditor-completion';sys.path.insert(0,str(ROOT/'scripts'))
from optimization_evidence import records
sha=lambda b:hashlib.sha256(b).hexdigest()
@lru_cache(maxsize=None)
def archived(folder):
 with tarfile.open(folder/'rss-raw.tar.gz','r:gz') as tar:
  return {m.name:tar.extractfile(m).read() for m in tar.getmembers() if m.isfile() and Path(m.name).name==m.name}
def read_raw(folder,name):
 path=folder/name
 return path.read_bytes() if path.is_file() else archived(folder)[name]

def summary(values):
 assert len(values)==31 and all(isinstance(v,(int,float)) and v>0 for v in values)
 return {'p50':statistics.median(values),'p95':sorted(values)[29],'min':min(values),'max':max(values)}

for stage in stages:
 folder=BASE/stage;ctx=json.loads((folder/'context.json').read_text());assert ctx['status']=='complete'
 assert sha((BASE/'procedures/measure-streaming.py.txt').read_bytes())==ctx['procedure_sha256']
 for name,digest in ctx['raw_sha256'].items():assert sha(read_raw(folder,name))==digest,(stage,name)
 parser=records((folder/'phase-timing.jsonl').read_text());assert len(parser)==8
 outputs=[json.loads(s) for s in (folder/'phase-outputs.jsonl').read_text().splitlines()];assert len(outputs)==8
 assert {r['case']:r['output'] for r in outputs}=={n:r['output'] for n,r in parser.items()}

 contexts[stage]=ctx
