package main
import("fmt";"image/color";"os";"runtime";progress "charm.land/bubbles/v2/progress";lip "charm.land/lipgloss/v2")
const preciseLinearSemantics="fresh premultiplied RGBA16-stop CIELAB configuration, complete 80-color canonical RGB8 vector and FNV64 checksum; include allocation, integer normalization and all output colors; no formatting or terminal I/O"
const preciseFrameSemantics="existing configured premultiplied RGBA16 progress at fraction 0.625; construct complete styled one-row frame and consume all UTF-8 symbols and foreground/background colors with FNV64; include frame allocation, formatting and rendering; no terminal I/O or animation"
func preciseColors(mode int)[]color.Color{alpha:=uint16(32768);if mode==1{alpha=0};return []color.Color{color.RGBA64{R:32768,G:8192,B:16384,A:alpha},color.RGBA64{R:65535,G:65535,B:65535,A:65535}}}
func preciseVector(stops []color.Color)[][3]uint8{values:=lip.Blend1D(80,stops...);out:=make([][3]uint8,len(values));for i,c:=range values{r,g,b,_:=c.RGBA();out[i]=[3]uint8{uint8(r>>8),uint8(g>>8),uint8(b>>8)}};return out}
func preciseChecksum(colors [][3]uint8)uint64{value:=uint64(14695981039346656037);for _,c:=range colors{for _,v:=range c{value=(value^uint64(v))*1099511628211}};return value}
func init(){if len(os.Args)<2||os.Args[1]!="--precise-colors"{return};for mode:=range 2{stops:=preciseColors(mode);values:=preciseVector(stops);output:="";for i,v:=range values{if i>0{output+=","};output+=fmt.Sprintf("%02x%02x%02x",v[0],v[1],v[2])};measure(fmt.Sprintf("precise/fresh_%d_80",mode),1,func()uint64{out:=preciseVector(stops);value:=preciseChecksum(out);runtime.KeepAlive(out);return value},output,preciseLinearSemantics)
 width:=40;if mode==1{width=80};m:=progress.New(progress.WithWidth(width),progress.WithColors(stops...));frame:=progressFrame(m,0.625);measure(fmt.Sprintf("precise/frame_%d_%d",mode,width),1,func()uint64{frame:=progressFrame(m,0.625);value:=progressChecksum(frame);runtime.KeepAlive(frame);return value},progressOutput(frame),preciseFrameSemantics)};os.Exit(0)}
