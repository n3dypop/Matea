sha=lambda b:hashlib.sha256(b).hexdigest()
@lru_cache(maxsize=None)
def archive(folder):
 with tarfile.open(folder/'rss-raw.tar.gz','r:gz') as tar:return {m.name:tar.extractfile(m).read() for m in tar.getmembers() if m.isfile() and Path(m.name).name==m.name}
def raw(folder,name):return (folder/name).read_bytes() if (folder/name).is_file() else archive(folder)[name]
def stats(samples):

 return {'p50':statistics.median(samples),'p95':sorted(samples)[29],'min':min(samples),'max':max(samples)}
