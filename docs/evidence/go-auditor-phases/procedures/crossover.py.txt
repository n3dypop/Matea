sha=lambda b:hashlib.sha256(b).hexdigest()
BINS={'original_88781a7':Path('/private/tmp/matea-go-auditor-before'),'contained_421feb3':Path('/private/tmp/matea-go-auditor-after'),'extracted_3df2002':Path('/private/tmp/matea-go-auditor-phase-pre-root-reuse'),'root_entry_reuse':Path('/private/tmp/matea-go-auditor-root-reuse')}
