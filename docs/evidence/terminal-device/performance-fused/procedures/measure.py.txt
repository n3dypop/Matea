 try:
  child=subprocess.run(command,stdin=slave,capture_output=True,env=environment,timeout=180)
  name=implementation+'-'+label
  (attempt/(name+'.jsonl')).write_bytes(child.stdout);(attempt/(name+'.stderr.txt')).write_bytes(child.stderr)
  after=termios.tcgetattr(slave);item.update(exit_code=child.returncode,stdout=name+'.jsonl',stderr=name+'.stderr.txt',after=portable(after),stdout_sha256=sha(child.stdout),stderr_sha256=sha(child.stderr));save()

  require(configured(after)==configured(before),'configured device fields changed: '+name)

   matches=re.findall(rb'^\s*(\d+)\s+maximum resident set size\s*$',child.stderr,re.M)

    require(r['calibration']['target_ns']==20000000,'changed calibration target')
