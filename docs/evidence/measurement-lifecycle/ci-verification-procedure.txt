import hashlib,json,subprocess
from pathlib import Path
ROOT=Path('/Users/misosoup/Developer/Codex/Project Matea/Matea')
ART=Path('/private/tmp/matea-measurement-ci-artifacts')
RUN=json.loads(Path('/private/tmp/matea-measurement-ci-run.json').read_bytes())
HEAD=RUN['headSha']
OUT=ROOT/'docs/evidence/measurement-lifecycle'
def sha(raw):return hashlib.sha256(raw).hexdigest()
def read(p):return json.loads(p.read_bytes())
def check(value,message):
 if not value:raise ValueError(message)
def sources(mapping):
 for name,want in mapping.items():
  raw=subprocess.check_output(['git','show',HEAD+':'+name],cwd=ROOT)
  check(sha(raw)==want,'source mismatch '+name)
def rows(path):return [json.loads(line) for line in path.read_bytes().splitlines()]
def measurement(folder,context_name,raw_name,count):
 context=read(folder/context_name);check(context['status']=='complete','incomplete '+str(folder))
 check(context['git_head']==HEAD,'head mismatch '+str(folder))
 raw=(folder/raw_name).read_bytes();check(sha(raw)==context['raw_sha256'],'raw mismatch '+str(folder))
 values=rows(folder/raw_name);check(len(values)==count,'case count '+str(folder))
 check(len({v['case'] for v in values})==count,'duplicate cases')
 for row in values:
  check(len(row['samples_ns'])==31 and all(type(n)is int and n>0 for n in row['samples_ns']),'invalid interval')
 return context,values
jobs=[j for j in RUN['jobs'] if j['name'] not in ['reference-api','reference-probes']]
check(len(jobs)==7 and all(j['conclusion']=='success' for j in jobs),'required CI jobs not all successful')
result={'schema':1,'run':RUN['url'],'source_head':HEAD,'all_7_required_jobs_passed':True,'hosts':{},'raw_artifact_sha256':{},'limits':['Whole driver execution, deterministic complete outputs and input/schema binding, not shared-runner performance comparisons.','Compiler captures cover two native all-feature profiles per host; existing twenty-profile API records are not recollected or closed.','Raw data stays independently hashed; retained log copies normalize line endings and redact machine directories.']}
all_outputs={}
for host in ['ubuntu','windows','macos']:
 folder=ART/f'measurement-drivers-{host}-latest'
 report=read(folder/'qualification.json')
 check(report['status']=='complete' and report['git_head']==HEAD,'qualification incomplete/head '+host)
 check(len(report['checks'])==18 and all(v['exit_code']==0 for v in report['checks']),'qualification checks '+host)
 check(len(report['source_sha256'])==147,'known input count')
 sources(report['source_sha256'])
 for item in report['checks']:check(sha((folder/item['log']).read_bytes())==item['log_sha256'],'log hash '+host+'/'+item['name'])
 auditors={}
 for lang,expected in [('rust',3),('go',4)]:
  cohorts={}
  for stage in ['before','after']:
   context,values=measurement(folder/f'auditor-{lang}',stage+'-context.json',stage+'.jsonl',expected)
   sources(context['source_sha256'])
   for row in values:check(type(row['cold_ns'])is int and row['cold_ns']>0,'missing cold auditor interval')
   synthetic=next(v for v in values if v['case']=='synthetic_10000')
   text=''.join(f'#[cfg(unix)] pub const ITEM_{n}: usize = {n};\n' for n in range(10000)) if lang=='rust' else 'package fixture\n'+''.join(f'const ITEM_{n} = {n}\n' for n in range(10000))
   check(synthetic['input']['generated_lib_sha256']==sha(text.encode()),'generated bytes hash '+lang)
   cohorts[stage]={v['case']:{key:v[key] for key in ['case','semantics','input','output_sha256','output_bytes','declarations']} for v in values}
  check(cohorts['before']==cohorts['after'],'auditor complete output changed '+lang)
  auditors[lang]={'cases_per_stage':expected,'positive_intervals_per_case':31,'cold_retained':True,'synthetic_utf8_lf_hash_checked':True,'outputs':cohorts['before']}
 context,values=measurement(folder/'remaining','context.json','current.jsonl',30)
 check(context['source_sha256']==report['source_sha256'],'remaining input snapshot')
 check({v['case'] for v in values if v['case'].startswith('examples/')}=={'examples/'+n for n in ['counter','list','form','dashboard','effects','motion','input','pagination','help','clocks','spinners','cursor','attributes','gradients','progress','external-widget']},'example cases')
 check(len(context['compiler_inputs'])==2,'native compiler captures missing')
 pins={v['name']:v['sha'] for v in read(ROOT/'docs/upstream-lock.json')}
 for ref,package in [('crossterm','crossterm'),('ratatui','ratatui-core')]:
  item=context['compiler_inputs']['compiler/extract-'+ref+'-all']
  check(item['sha']==pins[ref] and item['profile']=='all' and item['package']==package,'native capture identity')
  check(item['raw_bytes']>0 and item['items']>0 and item['edges']>0 and len(item['source_manifest_sha256'])==64,'native capture schema')
 reader=read(folder/'reader/context.json')
 check(reader['status']=='complete' and reader['git_head']==HEAD and reader['source_sha256']==report['source_sha256'],'reader snapshot')
 check([v['stage'] for v in reader['runs']]==['before','after','after','before'],'reader balanced order')
 outputs=[]
 for item in reader['runs']:
  check(sha((folder/'reader'/item['raw']).read_bytes())==item['raw_sha256'],'reader raw')
  vals=rows(folder/'reader'/item['raw']);check(len(vals)==1,'reader cases')
  row=vals[0];check(len(row['samples_ns'])==31 and all(type(n)is int and n>0 for n in row['samples_ns']),'reader intervals')
  outputs.append(row['output_sha256'])
 check(len(set(outputs))==1,'reader complete output mismatch')
 all_outputs[host]={v['case']:v['output_sha256'] for v in values if not v['case'].startswith('compiler/extract-') and v['case'] not in ['coverage/complete-current-ledger-cli','reporters/generate-complete-retained-comparison']}
 native=ART/f'native-drivers-{host}-latest'
 local=read(native/'local-validation.json');llvm=read(native/'quality/rust-execution-coverage.json')
 check(local['status']=='complete' and local['all_local_checks_passed']is True and all(c['exit_code']==0 for c in local['checks']),'native acceptance failed')
 check(local['source_sha256']==report['source_sha256']==llvm['input_sha256'],'native/LLVM input mismatch')
 check(llvm['status']=='complete','LLVM incomplete')
 uploaded=0
 for c in local['checks']:
  p=native/c['log']
  if p.exists():check(sha(p.read_bytes())==c['log_sha256'],'native log '+str(p));uploaded+=1
 result['hosts'][host]={'qualification_checks':18,'known_inputs':147,'source_matches_git':True,'auditors':auditors,'remaining_cases':30,'examples':16,'platform':report['platform'],'cross_host_equality_exclusions':['Two native compiler graphs are platform/feature-specific.','Python coverage CLI uses native stdout line endings; records retain raw bytes.','Full reporter includes current host/architecture/rustc/Go metadata; only per-host complete output determinism is claimed. Removed-metadata report equivalence is not independently reconstructed.'],'captured_inputs':context['compiler_inputs'],'balanced_reader_runs':4,'reader_output_sha256':outputs[0],'acceptance_checks':len(local['checks']),'uploaded_native_logs_hash_checked':uploaded,'llvm_source_files':len(llvm['files']),'llvm_compiler_instances':sum(len(f['instances']) for f in llvm['files'])}
 for p in sorted(folder.rglob('*')):
  if p.is_file():result['raw_artifact_sha256'][str(p.relative_to(ART))]=sha(p.read_bytes())
 for p in sorted(native.rglob('*')):
  if p.is_file():result['raw_artifact_sha256'][str(p.relative_to(ART))]=sha(p.read_bytes())
check(all_outputs['ubuntu']==all_outputs['macos']==all_outputs['windows'],'portable complete outputs mismatch (excluding native captures, OS text transport and current-host reporter metadata)')
result['remaining_cross_host_exact_output_cases']=len(all_outputs['ubuntu'])
result['all_known_input_sha256']=report['source_sha256']
OUT.mkdir(parents=True,exist_ok=True)
(OUT/'ci-source-bindings.json').write_text(json.dumps(result,indent=2)+'\n')
print('All seven jobs, three native hosts, 147 known inputs, 18 qualification checks, all auditor outputs/intervals, 30 remaining paths and four reader runs verified.')
