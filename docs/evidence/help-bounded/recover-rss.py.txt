"""Recover all completed timing cohorts; retry only sandbox-denied RSS children."""
import hashlib,json,os,subprocess,sys
from pathlib import Path
if not __debug__:raise RuntimeError('recovery refuses -O')
root=Path.cwd();base=root/'docs/evidence/help-bounded';sys.path.insert(0,str(root/'scripts'))
from measurement_run import MeasurementRun
from performance_records import read_records,read_probe_outputs,summary
sha=lambda b:hashlib.sha256(b).hexdigest()
prior=json.loads((base/'context.json').read_bytes());assert prior['status']=='failed' and prior['completed_cases']==174
attempt=base/prior['attempt'];original=read_records((attempt/'cases.jsonl').read_text());assert len(original)==174
assert sha((attempt/'cases.jsonl').read_bytes())==prior['partial_raw_sha256']
assert 'Operation not permitted' in (attempt/'rss-baseline-0.stderr').read_text()
for n,h in prior['root_input_sha256'].items():assert sha((root/n).read_bytes())==h==sha(subprocess.check_output(['git','show',prior['source_head']+':'+n]))
for n,h in prior['archive_sha256'].items():assert sha((base/n).read_bytes())==h
oracle=read_probe_outputs((base/'baseline-outputs.jsonl').read_text());source=read_probe_outputs((base/'source-outputs.jsonl').read_text());assert len(oracle)==25 and len(source)==8
rebuilt={};rows=[]
for cohort,order in enumerate(prior['orders'],1):
 for variant in order:
  path=attempt/f'cohort-{cohort}-{variant}.jsonl';assert not path.with_suffix('.stderr').read_bytes()
  records=read_records(path.read_text());expected=source if variant=='source' else oracle;assert set(records)==set(expected)
  for case,r in records.items():
   assert len(r['samples_ns'])==31 and r['calibration']['target_ns']==20_000_000
   for k in ['output','semantics','first_return_value']:assert r[k]==expected[case][k]
   key=f'cohort-{cohort}/{variant}/{case}';rebuilt[key]={**r,'case':key,'source_case':case,'cohort':cohort,'variant':variant};rows.append({'cohort':cohort,'variant':variant,'case':case,**summary(r)})
assert rebuilt==original
out=base/'rss-recovery';out.mkdir(exist_ok=True);rss=[]
binaries={'rust':Path('/private/tmp/matea-help-bounded-target/release/matea-help-bounded'),'source':Path('/private/tmp/matea-help-bounded/source-probe')}
assert {n:sha(p.read_bytes()) for n,p in binaries.items()}==prior['binary_sha256']
with MeasurementRun(out,root=root) as run:
 run.update(scope='Independently reconstructed completed ordinary cohorts plus separately permitted RSS children; ordinary timings are not repeated.',
  source_head=prior['source_head'],root_input_sha256=prior['root_input_sha256'],producer_context='../context.json',producer_context_sha256=sha((base/'context.json').read_bytes()),
  procedure_sha256=sha(Path(__file__).read_bytes()),binary_sha256=prior['binary_sha256'],summary=rows,
  negative='Original collector completed all 174 ordinary records, then /usr/bin/time -l failed because sandbox sysctl kern.clockrate was denied. That failed attempt and child logs remain unchanged.')
 for r in rebuilt.values():run.record(r)
 for variant in ['baseline','current','source']:
  for index in range(31):
   cmd=[str(binaries['source'])] if variant=='source' else [str(binaries['rust']),variant]
   result=subprocess.run(['/usr/bin/time','-l',*cmd],env={**os.environ,'MATEA_PROBE_OUTPUT_ONLY':'1'},stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=30)
   path=out/run.attempt/f'rss-{variant}-{index}.jsonl';path.write_bytes(result.stdout);path.with_suffix('.stderr').write_bytes(result.stderr)
   assert result.returncode==0,(variant,index,result.stderr)
   assert read_probe_outputs(result.stdout.decode())==(source if variant=='source' else oracle)
   values=[int(line.split()[0]) for line in result.stderr.decode().splitlines() if line.strip().endswith('maximum resident set size')];assert len(values)==1 and values[0]>0
   rss.append({'variant':variant,'index':index,'maximum_resident_set_bytes':values[0],'stdout_sha256':sha(result.stdout),'stderr_sha256':sha(result.stderr)})
 for n,h in prior['root_input_sha256'].items():assert sha((root/n).read_bytes())==h
 assert {n:sha(p.read_bytes()) for n,p in binaries.items()}==prior['binary_sha256']
 run.complete(timing_records=174,retained_intervals=5394,rss=rss)
print('Recovered 174 unchanged timing records / 5394 intervals; measured 93 RSS children',flush=True)
