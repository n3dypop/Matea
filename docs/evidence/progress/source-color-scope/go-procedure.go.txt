package main
import (
 "encoding/json"
 "fmt"
 "image/color"
 "os"
 progress "charm.land/bubbles/v2/progress"
 lip "charm.land/lipgloss/v2"
 "github.com/charmbracelet/x/ansi"
)
func emit(name string, c color.Color) {
 r,g,b,a:=c.RGBA()
 _=json.NewEncoder(os.Stdout).Encode(map[string]any{"case":name,"rgba16":[]uint32{r,g,b,a},"sgr":ansi.NewStyle().ForegroundColor(c).String()})
}
func main() {
 for _,v:=range []uint16{0,1,2,127,128,254,255,256,257,511,512,65534,65535} {
  emit(fmt.Sprintf("rgba64/%d",v),color.RGBA64{R:v,G:v,B:v,A:65535})
 }
 for _,steps:=range []int{2,3,6,17} {
  stops:=[]color.Color{color.RGBA{A:255},color.RGBA{R:1,G:1,B:1,A:255}}
  for i,c:=range lip.Blend1D(steps,stops...) {emit(fmt.Sprintf("low-gradient/%d/%d",steps,i),c)}
 }
 p:=progress.New(progress.WithWidth(3),progress.WithoutPercentage(),progress.WithFillCharacters('█','░'),progress.WithColors(color.RGBA{A:255},color.RGBA{R:1,G:1,B:1,A:255}))
 _=json.NewEncoder(os.Stdout).Encode(map[string]any{"case":"progress/low-gradient/3","raw":p.ViewAs(1)})
 for i,style:=range []lip.Style{lip.NewStyle(),lip.NewStyle().Border(lip.NormalBorder()).Padding(2).Margin(3).Width(60).Height(4),lip.NewStyle().MaxWidth(3),lip.NewStyle().Underline(true).UnderlineSpaces(false),lip.NewStyle().Strikethrough(true).StrikethroughSpaces(false)} {
  q:=progress.New(progress.WithWidth(12));q.PercentageStyle=style
  _=json.NewEncoder(os.Stdout).Encode(map[string]any{"case":fmt.Sprintf("percentage-style/%d",i),"raw":q.ViewAs(0.5),"plain":ansi.Strip(q.ViewAs(0.5))})
 }
}
