def fnv(text,x=None,y=None):
    value=14695981039346656037
    raw=text.encode('utf-8')
    if x is not None:raw+=x.to_bytes(8,'little')+y.to_bytes(8,'little')
    for byte in raw:value=((value^byte)*1099511628211)&((1<<64)-1)
    return value
def check_output(data):

            expected=fnv(text[5:],int(x[2:]),int(y[2:]))
        require(row['first_return_value']==expected,'independent complete consumption mismatch '+name)
