import hashlib,json,statistics,subprocess,sys
from pathlib import Path
ROOT=Path('/Users/misosoup/Developer/Codex/Project Matea/Matea')
sys.path.insert(0,str(ROOT/'scripts'))
from performance_records import parse_json,read_records,read_probe_outputs
BASE=ROOT/'docs/evidence/viewport-horizontal'
def require(value,message):
    if not value:raise ValueError(message)
def sha(data):return hashlib.sha256(data).hexdigest()
def read(path):return parse_json(path.read_bytes())
def audit(directory,flag):
    context=read(directory/'context.json')
    require(context['status']=='complete' and context['parameters']['mode']==flag,'incomplete/wrong corpus')
    require(context==read(directory/context['attempt']/'context.json'),'journal mismatch')
    require(sha((directory/context['raw']).read_bytes())==context['raw_sha256'],'manifest hash')
    require(sha((directory/'report.json').read_bytes())==context['report_sha256'],'report hash')
    for name,digest in context['source_sha256'].items():require(sha(subprocess.check_output(['git','show',context['source_head']+':'+name],cwd=ROOT))==digest,'Git source mismatch '+name)
    manifest=[parse_json(row) for row in (directory/context['raw']).read_text().splitlines()]
    require(len(manifest)==4 and [row['language'] for row in manifest]==['upstream','matea','matea','upstream'],'order changed')
    output={language:read_probe_outputs((directory/context['attempt']/(language+'-outputs.jsonl')).read_text()) for language in ['matea','upstream']}
    require(len(output['matea'])==18 and len(output['upstream'])==14,'output case count')
    samples=[]
    for row in manifest:
        path=directory/row['raw'];require(sha(path.read_bytes())==row['sha256'],'raw hash')
        data=read_records(path.read_text());require(data.keys()==output[row['language']].keys(),'case set changed')
        for name,item in data.items():
            require(all(item[key]==output[row['language']][name][key] for key in ['output','semantics','first_return_value']),'measured output changed')
            require(len(item['samples_ns'])==31 and item['calibration']['target_ns']==20_000_000,'calibration/sample contract')
        samples.append(data)
    return context,output,samples
report={'schema':1,'status':'complete_bounded_comparison','corpora':{},'limits':['Two finite 100-line ASCII/Unicode corpora, repeated and unique per-line numeric suffixes. Same tracked benchmark caller source and Go binary across before/after.','Separate reference/Matea/Matea/reference cohorts before/after, native macOS arm64 only. No affinity/frequency lock or RSS instrumentation; no causal or statistical claim for small differences.','Go constructor default features remain more extensive; Go View/plain adapter and Rust scene/surface/plain conversion are distinct internal work. No full viewport/module parity, all-high or maximum-speed verdict.','All cold calls, 31 positive batch means, calibration histories, requested allocation counters and adverse controls are retained. Rust layout requests, Go TotalAlloc and RSS are different measures.']}
for corpus,flag in [('repeated','--viewport-geometry'),('varied','--viewport-geometry-varied')]:
    before,old,old_runs=audit(BASE/('performance-before-'+corpus),flag)
    after,new,new_runs=audit(BASE/('performance-after-'+corpus),flag)
    require(old==new,'before/after complete outputs changed')
    require(before['binary_sha256']['upstream']==after['binary_sha256']['upstream'],'reference executable changed')
    require(before['external_input_sha256']==after['external_input_sha256'],'Go caller/module/checksum inputs changed')
    for path in ['tools/perf-probe/src/main.rs','tools/perf-probe/src/measurement.rs','tools/perf-upstream/main.go','tools/perf-upstream/main_test.go','Cargo.toml','Cargo.lock']:
        require(before['source_sha256'][path]==after['source_sha256'][path],'benchmark/dependency source changed '+path)
    changes={p for p in before['source_sha256'].keys()|after['source_sha256'].keys() if before['source_sha256'].get(p)!=after['source_sha256'].get(p)}
    expected={'.github/workflows/foundation.yml','crates/matea-widgets/src/lib.rs','crates/matea-widgets/src/viewport.rs','crates/matea-widgets/tests/viewport.rs','docs/quality-plan.json','scripts/check_reference_paths.py'}
    require(changes==expected,'unexpected changes '+str(changes^expected))
    rows=[]
    for name in old['matea']:
        pairs=[]
        for index in [1,2]:
            a,b=old_runs[index][name],new_runs[index][name]
            p50a,p50b=statistics.median(a['samples_ns']),statistics.median(b['samples_ns']);p95a,p95b=sorted(a['samples_ns'])[29],sorted(b['samples_ns'])[29]
            pairs.append({'before_p50_ns':p50a,'after_p50_ns':p50b,'p50_after_over_before':p50b/p50a,'before_p95_ns':p95a,'after_p95_ns':p95b,'p95_after_over_before':p95b/p95a,'before_allocations':a['allocations'],'after_allocations':b['allocations'],'before_requested_bytes':a['allocated_bytes'],'after_requested_bytes':b['allocated_bytes']})
        rows.append({'case':name,'comparison':'distinct Rust iterator control' if '/borrowed_' in name else 'matched Go/Rust consumer','complete_output_sha256':sha(new['matea'][name]['output'].encode()),'first_return_value':new['matea'][name]['first_return_value'],'forward_and_reverse':pairs})
    report['corpora'][corpus]={'before_source':before['source_head'],'after_source':after['source_head'],'matched_cases':14,'distinct_Rust_controls':4,'same_complete_outputs_before_after':True,'same_reference_binary':True,'same_benchmark_sources_and_lockfile':True,'changed_inputs':sorted(changes),'cases':rows}
(BASE/'optimized/performance-comparison.json').write_text(json.dumps(report,indent=2)+'\n')
print('Both corpora: 14 matched operations + 4 distinct controls, 16 raw runs, all source/output/hash/sample checks passed; adverse distributions retained.')
