"""Bind the committed clock reference assessment without relabeling old execution."""
import hashlib,json,subprocess,sys
from pathlib import Path
if not __debug__:raise RuntimeError('binding refuses -O')
root=Path.cwd();sys.path.insert(0,str(root/'scripts'))
from check_quality import execution_inputs,programs,validate_native,validate_coverage
sha=lambda b:hashlib.sha256(b).hexdigest()
head=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
native=json.loads((root/'docs/evidence/local-validation.json').read_bytes());coverage=json.loads((root/'docs/evidence/quality/rust-execution-coverage.json').read_bytes())
validate_native(native,root);validate_coverage(coverage,root)
actual={n:sha((root/n).read_bytes()) for n in sorted(execution_inputs(root))}
assert actual==native['source_sha256']==coverage['input_sha256'] and len(actual)==198 and len(programs(root))==156
for name,h in actual.items():assert sha(subprocess.check_output(['git','show',head+':'+name]))==h,name
ci=json.loads((root/'docs/evidence/terminal-device/ci-04e81bc/source-bindings.json').read_bytes())
assert sorted(n for n in actual if actual[n]!=ci['input_sha256'][n])==['docs/quality-plan.json']
clock=json.loads((root/'docs/evidence/clocks-exports/verification.json').read_bytes())
for name,h in clock['files_sha256'].items():assert sha((root/'docs/evidence/clocks-exports'/name).read_bytes())==h,name
for name,h in clock['root_input_sha256'].items():assert sha(subprocess.check_output(['git','show',clock['source_head']+':'+name]))==h,name
assert sorted(n for n in actual if actual[n]!=clock['root_input_sha256'][n])==['docs/quality-plan.json']
assert all(p['source_members']==43 for p in clock['profiles'].values()) and len({p['surface_sha256'] for p in clock['profiles'].values()})==1
for name,h in clock['current_tested_program_sha256'].items():assert actual[name]==h==ci['input_sha256'][name],name
old=json.loads(subprocess.check_output(['git','show',ci['source_head']+':docs/quality-plan.json']));new=json.loads((root/'docs/quality-plan.json').read_bytes())
assert old['high_threshold']==new['high_threshold']==75 and old['scope']==new['scope'] and len(old['units'])==len(new['units'])==44
allowed=set(ci['qualified_units']);promoted=[]
for a,b in zip(old['units'],new['units']):
 assert a['id']==b['id'] and a['files']==b['files']
 for dimension,value in b['criteria'].items():
  previous=a['criteria'][dimension]
  if value['score']==previous['score']:continue
  assert previous['score']==50 and value['score']==75
  if a['id']=='matea-widgets/clocks':assert dimension=='reference'
  else:assert a['id'] in allowed and (dimension=='portability' or (a['id']=='matea-terminal/device' and dimension=='behavior'))
  promoted.append([a['id'],dimension])
assert len(promoted)==8 and ['matea-widgets/clocks','reference'] in promoted
assert json.loads((root/'docs/coverage/dependency-frontier.json').read_bytes())['go_target_export_closure_reviewed'] is False
gate=json.loads((root/'docs/evidence/quality/final-gate-status.json').read_bytes());assert gate['below_high_axes']==49 and gate['pending_issue_reports']==822 and gate['f5']=='closed'
result={'schema':1,'status':'complete','tested_committed_inputs':head,'ci_source_head':ci['source_head'],'execution_inputs':198,'programs':156,'review_units':44,
 'current_inputs_match_native_and_llvm':True,'ci_different_inputs':['docs/quality-plan.json'],'promoted_assessments':promoted,
 'clock_surface_verification_sha256':sha((root/'docs/evidence/clocks-exports/verification.json').read_bytes()),
 'reviewed_metadata_changes':'Six portability assessments plus device behavior qualified against exact-source native/instrumented CI. Clock reference assessment separately reviewed against 43 raw/compiler members on three profiles and existing exact-source native/source-fixture CI. Same files, units, scope and threshold; no performance promotion.',
 'gate':{'below_high_axes':49,'pending_issue_reports':822,'go_target_export_closure_reviewed':False,'stage_a':'ongoing','stage_b':'not_started','f5':'closed'},
 'report_sha256':{n:sha((root/n).read_bytes()) for n in ['docs/evidence/local-validation.json','docs/evidence/quality/rust-execution-coverage.json','docs/evidence/quality/report.json','docs/evidence/quality/final-gate-status.json']},
 'procedure_sha256':sha(Path(__file__).read_bytes()),'limits':['CI and compiler probes remain bound to their actual inputs/revisions; later separately reviewed assessment metadata does not relabel executions.','Clock functional correspondence includes the explicit ADR 0019 improvements. Active timer end-to-end performance, full external API/dependency/target closure and universal/exhaustive correctness remain pending.']}
(root/'docs/evidence/clocks-exports/post-qualification-binding.json').write_text(json.dumps(result,indent=2)+'\n');print('Bound',head,'198 inputs; 49 unmet axes; F5 closed')
