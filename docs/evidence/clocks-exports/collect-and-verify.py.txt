 repo=root.parent/'references'/ref
 raw=subprocess.check_output(['git','-C',str(repo),'show',pins[ref]['sha']+':'+name])
 assert raw==(repo/name).read_bytes(),(ref,name)
 reference_files[ref+'/'+name]={'reference_sha':pins[ref]['sha'],'sha256':sha(raw)}
profiles={};previous=None
for os,arch in [('darwin','arm64'),('linux','amd64'),('windows','amd64')]:

 assert not Path('/private/tmp/matea-clock-exports',os+'.log').read_bytes()
 shutil.copy2(p,base/p.name);shutil.copy2(Path('/private/tmp/matea-clock-exports',os+'.log'),base/(os+'.log'))
 surfaces=[]
 for group in r['profiles']:
  package=group['Package'].rsplit('/',1)[1];source=package+'/'+package+'.go'
  assert group['SourceFiles']==[package+'.go']
  assert Path(group['Directory']).resolve()==(root.parent/'references/bubbles'/package).resolve()

  for member in group['Members']:
   kind=member['Kind'];symbol=member['Symbol']
   if kind=='method':
    kind='function'
    if member['Receivers']==['pointer']:symbol='*'+symbol
    else:assert member['Receivers']==['value','pointer']
   key=(symbol,kind);assert key not in compiled;compiled[key]=member
  raw={(d['symbol'],d['kind']):d for d in inventory['declarations'] if d['file']==source and d['kind']!='benchmark'}
  assert set(compiled)==set(raw) and len(raw)=={'stopwatch':21,'timer':22}[package]
  capability=next(c for c in capabilities if c['id']=='widgets/'+{'stopwatch':'stopwatch','timer':'countdown'}[package])
  assert capability['status']=='implemented' and set(capability['reference_ids'])=={d['id'] for d in raw.values()}
  surfaces.append({'package':package,'members':[{'symbol':key[0],'kind':key[1],'reference_id':raw[key]['id'],'signature':compiled[key]['Type'],'receivers':compiled[key]['Receivers']} for key in sorted(raw)]})
 assert previous is None or previous==surfaces
 previous=surfaces;profiles[os]={'goarch':arch,'source_members':sum(len(s['members']) for s in surfaces),'surface_sha256':sha(json.dumps(surfaces,sort_keys=True).encode())}
