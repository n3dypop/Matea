use std::time::Duration;
use matea_render::Text;
pub struct ClockText(pub Duration);
struct DurationText {
    bytes: [u8; 40],
    start: usize,
}
impl DurationText {
    #[inline]
    fn prepend(&mut self, text: &[u8]) {
        let start = self.start - text.len();
        self.bytes[start..self.start].copy_from_slice(text);
        self.start = start;
    }
    #[inline]
    fn digit(&mut self, value: u8) {
        self.start -= 1;
        self.bytes[self.start] = b'0' + value;
    }
    #[inline]
    fn integer(&mut self, mut value: u64) {
        const PAIRS: &[u8;200] = b"00010203040506070809101112131415161718192021222324252627282930313233343536373839404142434445464748495051525354555657585960616263646566676869707172737475767778798081828384858687888990919293949596979899";
        while value >= 100 {
            let pair = (value % 100) as usize * 2;
            self.prepend(&PAIRS[pair..pair+2]);
            value /= 100;
        }
        if value < 10 {
            self.digit(value as u8);
        } else {
            let pair = value as usize * 2;
            self.prepend(&PAIRS[pair..pair+2]);
        }
    }
    #[inline]
    fn fraction(&mut self, mut value: u32, mut width: usize) {
        if value == 0 {
            return;
        }
        while value % 10 == 0 {
            value /= 10;
            width -= 1;
        }
        let target = self.start - width;
        self.integer(u64::from(value));
        while self.start > target {
            self.digit(0);
        }
        self.prepend(b".");
    }
    #[inline]
    fn as_str(&self) -> &str {
        std::str::from_utf8(&self.bytes[self.start..])
            .expect("clock text uses digits and static UTF-8 units")
    }
}
impl ClockText {
    /// Owned immutable display text, ready for structured views. The existing
    /// bounded formatter feeds Text directly, avoiding a formatting-arguments
    /// adapter. Short text may use inline storage; allocation is not an API
    /// guarantee. Standard Display width/precision options remain separate.
    ///
    /// ```
    /// use matea_widgets::ClockText;
    /// use std::time::Duration;
    /// let text = ClockText(Duration::from_nanos(1500)).into_text();
    /// assert_eq!(text.as_str(), "1.5µs");
    /// ```
    #[inline]
    pub fn into_text(self) -> Text {
        if self.0.is_zero() {
            Text::from("0s")
        } else {
            Text::from(self.formatted().as_str())
        }
    }
    /// One exact-size owned string allocation; no growing intermediate String.
    #[inline]
    pub fn into_string(self) -> String {
        if self.0.is_zero() {
            return "0s".to_owned();
        }
        String::from(self.formatted().as_str())
    }
    #[inline]
    fn formatted(self) -> DurationText {
        // Duration::MAX needs at most 33 bytes. Fill backward to avoid separate
        // integer/fraction arrays, scans and copies. Only complete static UTF-8
        // units and ASCII digits enter the final checked string slice.
        let mut out = DurationText {
            bytes: [0; 40],
            start: 40,
        };
        let seconds = self.0.as_secs();
        let nanos = self.0.subsec_nanos();
        if seconds == 0 {
            if nanos == 0 {
                out.prepend(b"0s");
            } else if nanos < 1_000 {
                out.prepend(b"ns");
                out.integer(u64::from(nanos));
            } else if nanos < 1_000_000 {
                out.prepend("µs".as_bytes());
                out.fraction(nanos % 1_000, 3);
                out.integer(u64::from(nanos / 1_000));
            } else {
                out.prepend(b"ms");
                out.fraction(nanos % 1_000_000, 6);
                out.integer(u64::from(nanos / 1_000_000));
            }
            return out;
        }
        out.prepend(b"s");
        out.fraction(nanos, 9);
        out.integer(seconds % 60);
        let hours = seconds / 3_600;
        let minutes = seconds / 60 % 60;
        if hours > 0 || minutes > 0 {
            out.prepend(b"m");
            out.integer(minutes);
        }
        if hours > 0 {
            out.prepend(b"h");
            out.integer(hours);
        }
        out
    }
}

pub fn owned(value:Duration)->Text{ClockText(value).into_text()}
